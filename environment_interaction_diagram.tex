% Environment Interaction Diagram for inclusion in main document
\begin{figure}[htbp]
\centering
\begin{tikzpicture}[scale=0.8, transform shape,
    % Node styles
    host/.style={circle, draw=blue!70, fill=blue!10, minimum size=6mm, font=\tiny},
    decoy/.style={circle, draw=red!70, fill=red!10, minimum size=6mm, font=\tiny, dashed},
    compromised/.style={circle, draw=red, fill=red!30, minimum size=6mm, font=\tiny},
    subnet/.style={rounded rectangle, draw=gray!70, fill=gray!10, minimum width=20mm, minimum height=12mm, font=\scriptsize},
    agent/.style={rectangle, draw=black, fill=yellow!20, minimum width=12mm, minimum height=6mm, font=\scriptsize, rounded corners=2pt},
    action/.style={->, thick, blue!70},
    attack/.style={->, thick, red!70, dashed},
    observation/.style={<->, thick, purple!70},
    reward/.style={<->, thick, orange!70}
]

% Network topology
\node[subnet] (subnet1) at (-2.5, 1) {\small Subnet $N_1$};
\node[subnet] (subnet2) at (0, 1) {\small Subnet $N_2$};
\node[subnet] (subnet3) at (2.5, 1) {\small Subnet $N_3$};

% Sample hosts
\node[host] (h1) at (-2.8, 1.3) {$h_1$};
\node[decoy] (d1) at (-2.2, 1.3) {$d_1$};
\node[compromised] (h2) at (0, 1.3) {$h_2$};
\node[host] (h3) at (2.5, 1.3) {$h_3$};

% Agents
\node[agent] (blue_agent) at (-3, -1) {\textbf{Blue Agent}\\Defender};
\node[agent] (red_agent) at (3, -1) {\textbf{Red Agent}\\Fixed Adversary};

% State and action representations
\node[draw=blue!50, fill=blue!5, align=left, font=\tiny, minimum width=25mm] (blue_obs) at (-3, -2.2) {
    \textbf{Observations:}\\
    Current alerts\\
    Alert history\\
    Decoy count
};

\node[draw=blue!50, fill=blue!5, align=left, font=\tiny, minimum width=25mm] (blue_act) at (-1, -1) {
    \textbf{Actions:}\\
    Deploy decoy\\
    Remove decoy\\
    Do nothing
};

\node[draw=red!50, fill=red!5, align=left, font=\tiny, minimum width=25mm] (red_act) at (1, -1) {
    \textbf{Actions:}\\
    Scan network\\
    Exploit hosts\\
    Lateral movement
};

% Interaction flows
\draw[action] (blue_agent) -- (blue_act);
\draw[action] (blue_act) to[bend left=15] (subnet1);
\draw[attack] (red_agent) -- (red_act);
\draw[attack] (red_act) to[bend right=15] (h2);

% Observation flow
\draw[observation] (subnet2) to[bend left=20] (blue_obs);
\draw[observation] (blue_obs) -- (blue_agent);

% Reward indication
\node[draw=orange!70, fill=orange!10, font=\tiny, align=center] (reward) at (0, -2.5) {
    \textbf{Immediate Rewards}\\
    $R_t^{(b)} = f(\text{actions}, \text{outcomes})$
};

\draw[reward] (blue_agent) to[bend right=15] (reward);
\draw[reward] (red_agent) to[bend left=15] (reward);

% Timestep flow
\node[draw=purple!70, fill=purple!10, font=\scriptsize] (timestep) at (0, 2) {Timestep $t$};
\draw[->, purple, thick] (red_agent) to[bend left=30] node[midway, above, font=\tiny] {1. Red acts} (timestep);
\draw[->, purple, thick] (timestep) to[bend left=30] node[midway, above, font=\tiny] {2. Blue observes \& acts} (blue_agent);

% Title
\node[above=5mm] at (timestep.north) {\textbf{Cyberwheel Multi-Agent Environment}};

\end{tikzpicture}
\caption{Cyberwheel environment showing multi-agent interaction between blue defender and red adversary. The blue agent deploys/removes decoys based on network observations, while the fixed red agent executes predetermined attack strategies. Both agents receive immediate rewards based on action outcomes.}
\label{fig:cyberwheel_environment}
\end{figure}
